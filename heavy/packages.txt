# What FerroOS Heavy ships on top of Debian's smallest base (debootstrap's
# "minbase"), one package a line. Kept short on purpose: every package is
# more to update and to trust, and everything else is one `apt install` away.

# Starting up: the kernel, init, and the pieces that unlock an encrypted
# disk at boot.
linux-image-amd64
systemd-sysv
udev
dbus
libpam-systemd
initramfs-tools
zstd
cryptsetup
cryptsetup-initramfs

# Running straight from the installation medium.
live-boot

# Secure Boot: Debian's signed boot loader.
grub-efi-amd64-signed
grub-efi-amd64-bin
shim-signed
efibootmgr
mokutil

# The duress PIN's slow, salted hash (installer, logon, disk unlock).
argon2

# What the installer uses, so installing needs no network.
gdisk
parted
dosfstools
e2fsprogs
squashfs-tools

# The display layer under our desktop, and the terminal (LXTerminal).
xserver-xorg-core
xserver-xorg-input-libinput
xserver-xorg-video-amdgpu
xserver-xorg-video-nouveau
xserver-xorg-video-fbdev
xserver-xorg-video-vesa
xinit
xauth
x11-xserver-utils
picom
xfonts-base
lxterminal
zenity

# Networking: wired and Wi-Fi, encrypted DNS, a firewall.
network-manager
iproute2
wpasupplicant
wireless-regdb
systemd-resolved
nftables
ca-certificates
# The right time: certificates (encrypted DNS, HTTPS) are checked against it.
systemd-timesyncd
# Time zones: without them every clock shows UTC (Date & Time sets one).
tzdata

# Firmware the common network cards and graphics won't work without. (Not free
# software; from Debian's non-free-firmware section.)
firmware-linux-free
firmware-realtek
firmware-iwlwifi
firmware-mediatek
firmware-atheros
firmware-brcm80211
firmware-amd-graphics
firmware-misc-nonfree
firmware-sof-signed

# Processor makers' fixes for their chips' security flaws, loaded at every
# start. (Also from non-free-firmware.)
intel-microcode
amd64-microcode

# Sound, hardening and administration. unattended-upgrades installs
# Debian's security updates daily (harden.sh limits it to those).
unattended-upgrades
alsa-utils
apparmor
sudo
procps
pciutils
usbutils
kbd
less
nano
curl
wget

# Preinstalled productivity & media tools
mpv
mpv-mpris
geany
# Metadata: what pictures and videos carry, to see, change or strip (Info).
libimage-exiftool-perl

# Removable storage, automounting & filesystems. polkitd is what grants
# (or refuses) mounting: udisks2 only recommends it, and without it every
# mount is refused, so no other drive or USB stick ever appears.
polkitd
udisks2
ntfs-3g
exfatprogs
# Programs installed into C:\Programs get a writable layer over their
# folder (ferro-installer), through bubblewrap.
bubblewrap

# Desktop integration, D-Bus session & XDG protocol handling
xdg-utils
dbus-x11

# Hardware 3D graphics & OpenGL/Vulkan acceleration
libgl1-mesa-dri
mesa-vulkan-drivers

# Crisp TrueType fonts for browsers, Electron and GUI apps
fonts-dejavu-core
fonts-liberation

# Archive, clipboard & file inspection utilities
unzip
zip
xz-utils
file
xclip

# Tor & Pluggable Transports (switched on and off by ferro-tor)
tor
obfs4proxy
snowflake-client

# Network tools for the terminal: ping, nslookup/dig/host, traceroute,
# ifconfig/netstat, whois, nc.
iputils-ping
iputils-tracepath
bind9-dnsutils
traceroute
net-tools
whois
netcat-openbsd
bash-completion
